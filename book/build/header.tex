% ---- headings (redefined with \@startsection, which the tagging code hooks) ----
% The chapter number is part of the heading text ("Chapter 2: ..."), so the opener splits the text at
% the first colon; the heading is one paragraph and keeps its full text.
\ExplSyntaxOn
\tl_new:N \l__sdv_kind_tl
\tl_new:N \l__sdv_num_tl
\tl_new:N \l__sdv_title_tl
\tl_new:N \l__sdv_whole_tl
\seq_new:N \l__sdv_seq
\cs_new_protected:Npn \sdv_split_heading:n #1
  {
    \tl_clear:N \l__sdv_kind_tl
    \tl_clear:N \l__sdv_num_tl
    \tl_set:Nn \l__sdv_title_tl {#1}
    \regex_extract_once:nnNT { \A (Chapter|Part|Appendix)\ ([0-9A-Z]+)\:\  (.*) } {#1} \l__sdv_seq
      {
        \seq_pop_left:NN \l__sdv_seq \l__sdv_whole_tl
        \seq_pop_left:NN \l__sdv_seq \l__sdv_kind_tl
        \seq_pop_left:NN \l__sdv_seq \l__sdv_num_tl
        \seq_pop_left:NN \l__sdv_seq \l__sdv_title_tl
      }
  }
\NewDocumentCommand \sdvsplitheading { m } { \sdv_split_heading:n {#1} }
\NewDocumentCommand \sdvheadkind { } { \tl_use:N \l__sdv_kind_tl }
\NewDocumentCommand \sdvheadnum { } { \tl_use:N \l__sdv_num_tl }
\NewDocumentCommand \sdvheadtitle { } { \tl_use:N \l__sdv_title_tl }
